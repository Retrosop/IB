from math import gcd

# Функция для нахождения мультипликативной обратной величины
def mod_inverse(k, n):
    for i in range(1, n):
        if (k * i) % n == 1:
            return i
    raise ValueError(f"Мультипликативная обратная величина для {k} по модулю {n} не существует.")

# Функция шифрования
def affine_encrypt(text, k, a, n=26):
    encrypted_text = ""
    for char in text:
        if char.isalpha():
            # Преобразуем символ в позицию в алфавите (0-25)
            is_upper = char.isupper()
            x = ord(char.upper()) - ord('A')
            # Применяем формулу шифрования
            y = (k * x + a) % n
            # Преобразуем обратно в символ
            encrypted_char = chr(y + ord('A'))
            encrypted_text += encrypted_char if is_upper else encrypted_char.lower()
        else:
            encrypted_text += char  # Не шифруем неалфавитные символы
    return encrypted_text

# Функция дешифрования
def affine_decrypt(encrypted_text, k, a, n=26):
    decrypted_text = ""
    k_inverse = mod_inverse(k, n)  # Находим мультипликативную обратную величину
    for char in encrypted_text:
        if char.isalpha():
            # Преобразуем символ в позицию в алфавите (0-25)
            is_upper = char.isupper()
            y = ord(char.upper()) - ord('A')
            # Применяем формулу дешифрования
            x = (k_inverse * (y - a)) % n
            # Преобразуем обратно в символ
            decrypted_char = chr(x + ord('A'))
            decrypted_text += decrypted_char if is_upper else decrypted_char.lower()
        else:
            decrypted_text += char  # Не дешифруем неалфавитные символы
    return decrypted_text.strip()
