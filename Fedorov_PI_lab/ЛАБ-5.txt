import sympy

def generate_rsa_keypair():
    # Шаг 1: Выбор двух простых чисел
    p = sympy.randprime(100, 200)  # Генерация случайного простого числа в диапазоне
    q = sympy.randprime(100, 200)  # Генерация второго простого числа

    # Шаг 2: Вычисление n
    n = p * q

    # Шаг 3: Вычисление функции Эйлера
    phi_n = (p - 1) * (q - 1)

    # Шаг 4: Выбор открытой экспоненты e
    e = 65537  # Часто используемое значение для e

    # Проверка, что e взаимно просто с phi(n)
    if sympy.gcd(e, phi_n) != 1:
        raise ValueError("Выбранное значение e не является взаимно простым с phi(n)")

    # Шаг 5: Открытый ключ
    public_key = (e, n)

    return public_key, p, q

# Генерация открытого ключа
public_key, p, q = generate_rsa_keypair()
print(f"Открытый ключ: {public_key}")
print(f"Простые числа p: {p}, q: {q}")
