import random

def mod_exp(base, exp, mod):
    """Возводит base в степень exp по модулю mod."""
    return pow(base, exp, mod)

def generate_keys(p, g, a):
    """Генерирует открытый и закрытый ключи."""
    private_key = a
    y = mod_exp(g, private_key, p)
    return private_key, y

def encrypt(p, g, y, message, k):
    """Шифрует сообщение с помощью открытого ключа."""
    Y1 = mod_exp(g, k, p)
    Y2 = (mod_exp(y, k, p) ^ message)  # Используем XOR для шифрования
    return Y1, Y2

def decrypt(Y1, Y2, private_key, p):
    """Дешифрует сообщение с помощью закрытого ключа."""
    s = mod_exp(Y1, private_key, p)
    message = Y2 ^ s  # Используем XOR для расшифровки
    return message

# Параметры
P = 11  # Модуль
G = 8   # Генератор
A = 3   # Секретный ключ

# Генерация ключей
private_key, Y = generate_keys(P, G, A)
print(f"Закрытый ключ: {private_key}, Открытый ключ: {Y}")

# Шифрование сообщения
M = 5  # Сообщение
K = 9  # Случайное число
Y1, Y2 = encrypt(P, G, Y, M, K)
print(f"Зашифрованное сообщение: (Y1, Y2) = ({Y1}, {Y2})")

# Дешифрование сообщения
decrypted_message = decrypt(Y1, Y2, private_key, P)
print(f"Расшифрованное сообщение: {decrypted_message}")
