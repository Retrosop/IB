def additive_sensor(m, Y0, Y1, n):
    """
    Генерация последовательности с использованием аддитивного датчика.
    
    :param m: Модуль (максимальное значение).
    :param Y0: Начальное значение (seed).
    :param Y1: Второе значение последовательности.
    :param n: Количество значений, которые нужно сгенерировать.
    :return: Список из n значений последовательности.
    """
    # Инициализация первых двух значений
    sequence = [Y0, Y1]
    
    # Генерация последовательности
    for i in range(2, n):
        next_value = (sequence[i - 1] + sequence[i - 2]) % m
        sequence.append(next_value)
    
    return sequence


# Пример использования
if __name__ == "__main__":
    m = 4096
    Y0 = 4003
    Y1 = 59
    n = 10  # Количество значений, которые нужно сгенерировать

    result = additive_sensor(m, Y0, Y1, n)
    print("Сгенерированная последовательность:", result)
